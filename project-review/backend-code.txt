import { Request, Response, NextFunction } from "express";

import cloudinary from "../config/cloudinary";

import User from "../models/User";

const cookieOptions = {
  httpOnly: true,
  secure: process.env.NODE_ENV === "production",
  sameSite: "lax" as const,
  path: "/",
  maxAge: 24 * 60 * 60 * 1000,
};

const uploadImage = (
  buffer: Buffer
): Promise<{ url: string; publicId: string }> =>
  new Promise((resolve, reject) => {
    const stream = cloudinary.uploader.upload_stream(
      {
        folder: "fittrack/profile-images",
        resource_type: "image",
      },
      (error, result) => {
        if (error || !result) {
          reject(error ?? new Error("Image upload failed"));
          return;
        }

        resolve({
          url: result.secure_url,
          publicId: result.public_id,
        });
      }
    );

    stream.end(buffer);
  });

const removeImage = async (publicId?: string): Promise<void> => {
  if (!publicId) return;

  try {
    await cloudinary.uploader.destroy(publicId);
  } catch (error) {
    console.error("Cloudinary cleanup failed:", error);
  }
};

const safeUser = (user: {
  _id: unknown;
  name: string;
  email: string;
  phone?: string;
  role: string;
  profileImage?: { url: string; publicId: string };
}) => ({
  id: String(user._id),
  name: user.name,
  email: user.email,
  phone: user.phone ?? null,
  role: user.role,
  profileImage: user.profileImage?.url ?? null,
});

// Register a new member.
export const register = async (
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> => {
  let uploadedPublicId: string | undefined;

  try {
    const parsed = registerSchema.safeParse(req.body);

    if (!parsed.success) {
      res.status(400).json({
        success: false,
        message: "Invalid registration details.",
        errors: parsed.error.issues,
      });
      return;
    }

    if (!req.file) {
      res.status(400).json({
        success: false,
        message: "Profile picture is required.",
      });
      return;
    }

    const { name, phone, email, password } = parsed.data;

    const existingUser = await User.findOne({ email });

    if (existingUser) {
      res.status(409).json({
        success: false,
        message: "An account with this email already exists.",
      });
      return;
    }

    const profileImage = await uploadImage(req.file.buffer);
    uploadedPublicId = profileImage.publicId;

    const passwordHash = await bcrypt.hash(password, 12);

    // Public registration must never assign an administrator role.
    const user = await User.create({
      name,
      email,
      phone,
      passwordHash,
      role: "MEMBER",
      profileImage,
    });

    const token = createToken({
      userId: String(user._id),
      role: user.role,
    });

    res.cookie("fittrack_token", token, cookieOptions);

    res.status(201).json({
      success: true,
      message: "Registration successful.",
      user: safeUser(user),
    });
  } catch (error) {
    // Clean up the uploaded image if registration fails.
    await removeImage(uploadedPublicId);

    // Handle duplicate email errors, including concurrent registrations.
    if (
      typeof error === "object" &&
      error !== null &&
      "code" in error &&
      error.code === 11000
    ) {
      res.status(409).json({
        success: false,
        message: "An account with this email already exists.",
      });
      return;
    }

    next(error);
  }
};

// Log in an existing user.
export const login = async (
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> => {
  try {
    const parsed = loginSchema.safeParse(req.body);

    if (!parsed.success) {
      res.status(400).json({
        success: false,
        message: "Invalid login details.",
      });
      return;
    }

    const user = await User.findOne({
      email: parsed.data.email,
    }).select("+passwordHash");

    if (!user) {
      res.status(401).json({
        success: false,
        message: "Invalid email or password.",
      });
      return;
    }

    const passwordMatches = await bcrypt.compare(
      parsed.data.password,
      user.passwordHash
    );

    if (!passwordMatches || !user.isActive) {
      res.status(401).json({
        success: false,
        message: "Invalid email or password.",
      });
      return;
    }

    const token = createToken({
      userId: String(user._id),
      role: user.role,
    });

    res.cookie("fittrack_token", token, cookieOptions);

    res.status(200).json({
      success: true,
      message: "Login successful.",
      user: safeUser(user),
    });
  } catch (error) {
    next(error);
  }
};

// Update the authenticated user's profile picture.
export const updateProfileImage = async (
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> => {
  let uploadedPublicId: string | undefined;

  try {
    if (!req.file) {
      res.status(400).json({
        success: false,
        message: "Profile picture is required.",
      });
      return;
    }

    const user = await User.findById(req.user?.userId);

    if (!user) {
      res.status(404).json({
        success: false,
        message: "User not found.",
      });
      return;
    }

    if (!user.isActive) {
      res.status(401).json({
        success: false,
        message: "Your account is inactive.",
      });
      return;
    }

    const newImage = await uploadImage(req.file.buffer);
    uploadedPublicId = newImage.publicId;

    const oldPublicId = user.profileImage?.publicId;

    user.profileImage = newImage;
    await user.save();

    // The database now references the new image.
    uploadedPublicId = undefined;

    // Delete the old image only after the save succeeds.
    if (oldPublicId && oldPublicId !== newImage.publicId) {
      await removeImage(oldPublicId);
    }

    res.status(200).json({
      success: true,
      message: "Profile picture updated successfully.",
      user: safeUser(user),
    });
  } catch (error) {
    await removeImage(uploadedPublicId);
    next(error);
  }
};

import { Request, Response, NextFunction } from "express";
import mongoose from "mongoose";

import {
  createMembershipSchema,
  cancelMembershipSchema,
  renewMembershipSchema,
} from "../validators/membershipValidator";

// Calculate the correct status using membership dates.
const getCurrentStatus = (
  startDate: Date,
  expiryDate: Date
): "UPCOMING" | "ACTIVE" | "EXPIRED" => {
  const now = new Date();

  if (expiryDate <= now) {
    return "EXPIRED";
  }

  if (startDate > now) {
    return "UPCOMING";
  }

  return "ACTIVE";
};

// Refresh date-based membership statuses in the database.
const refreshMembershipStatuses = async (): Promise<void> => {
  const now = new Date();

  await Membership.updateMany(
    {
      status: { $in: ["ACTIVE", "UPCOMING"] },
      expiryDate: { $lte: now },
    },
    { $set: { status: "EXPIRED" } }
  );

  await Membership.updateMany(
    {
      status: "UPCOMING",
      startDate: { $lte: now },
      expiryDate: { $gt: now },
    },
    { $set: { status: "ACTIVE" } }
  );
};

// POST /api/memberships
// Admin assigns a membership plan to a member.
export const createMembership = async (
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> => {
  try {
    const validation = createMembershipSchema.safeParse(req.body);

    if (!validation.success) {
      res.status(400).json({
        success: false,
        message: "Validation failed",
        errors: validation.error.flatten(),
      });
      return;
    }

    const { memberId, planId, startDate } = validation.data;

    const currentUser = req.user;

    if (!currentUser) {
      res.status(401).json({
        success: false,
        message: "Authentication required",
      });
      return;
    }

    // Only an active MEMBER account can receive a membership.
    const member = await User.findOne({
      _id: memberId,
      role: "MEMBER",
      isActive: true,
    });

    if (!member) {
      res.status(404).json({
        success: false,
        message: "Active member not found",
      });
      return;
    }

    // The selected membership plan must be active.
    const plan = await MembershipPlan.findOne({
      _id: planId,
      isActive: true,
    });

    if (!plan) {
      res.status(404).json({
        success: false,
        message: "Active membership plan not found",
      });
      return;
    }

    const membershipStartDate = startDate
      ? new Date(startDate)
      : new Date();

    if (Number.isNaN(membershipStartDate.getTime())) {
      res.status(400).json({
        success: false,
        message: "Invalid membership start date",
      });
      return;
    }

    const expiryDate = new Date(membershipStartDate);

    expiryDate.setDate(
      expiryDate.getDate() + plan.durationDays
    );

    if (
      Number.isNaN(expiryDate.getTime()) ||
      expiryDate <= membershipStartDate
    ) {
      res.status(400).json({
        success: false,
        message: "Invalid membership dates",
      });
      return;
    }

    // Prevent overlapping active or upcoming memberships.
    const existingMembership = await Membership.findOne({
      memberId: member._id,
      status: { $in: ["ACTIVE", "UPCOMING"] },
      startDate: { $lt: expiryDate },
      expiryDate: { $gt: membershipStartDate },
    });

    if (existingMembership) {
      res.status(409).json({
        success: false,
        message:
          "This member already has a membership covering the requested period",
      });
      return;
    }

    const membership = await Membership.create({
      memberId: member._id,
      planId: plan._id,
      startDate: membershipStartDate,
      expiryDate,
      priceCharged: plan.price,
      status: getCurrentStatus(
        membershipStartDate,
        expiryDate
      ),
      assignedBy: new mongoose.Types.ObjectId(
        currentUser.userId
      ),
    });

    res.status(201).json({
      success: true,
      message: "Membership assigned successfully",
      data: membership,
    });
  } catch (error) {
    next(error);
  }
};

// GET /api/memberships/me
// Member views only their own memberships.
export const getMyMemberships = async (
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> => {
  try {
    const currentUser = req.user;

    if (!currentUser) {
      res.status(401).json({
        success: false,
        message: "Authentication required",
      });
      return;
    }

    const memberships = await Membership.find({
      memberId: currentUser.userId,
    })
      .populate("planId", "name description durationDays price")
      .sort({ createdAt: -1 });

    // Refresh statuses without changing cancellations.
    for (const membership of memberships) {
      if (membership.status === "CANCELLED") {
        continue;
      }

      const currentStatus = getCurrentStatus(
        membership.startDate,
        membership.expiryDate
      );

      if (membership.status !== currentStatus) {
        membership.status = currentStatus;
        await membership.save();
      }
    }

    res.status(200).json({
      success: true,
      count: memberships.length,
      data: memberships,
    });
  } catch (error) {
    next(error);
  }
};

// GET /api/memberships/:id
// Admin can view any membership.
// Member can view only their own membership.
export const getMembershipById = async (
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> => {
  try {
    const { id } = req.params;

    if (!mongoose.isValidObjectId(id)) {
      res.status(400).json({
        success: false,
        message: "Invalid membership ID",
      });
      return;
    }

    const membership = await Membership.findById(id)
      .populate("memberId", "name email phone")
      .populate(
        "planId",
        "name description durationDays price"
      );

    if (!membership) {
      res.status(404).json({
        success: false,
        message: "Membership not found",
      });
      return;
    }

    const currentUser = req.user;

    if (!currentUser) {
      res.status(401).json({
        success: false,
        message: "Authentication required",
      });
      return;
    }

    const isAdmin = currentUser.role === "ADMIN";

    // Safely obtain the member ID whether the reference
    // is populated or is still a MongoDB ObjectId.
    const memberReference = membership.memberId as unknown as {
      _id?: { toString(): string };
      toString(): string;
    };

    const membershipMemberId =
      memberReference?._id?.toString() ??
      memberReference?.toString();

    const belongsToMember =
      membershipMemberId === currentUser.userId;

    if (!isAdmin && !belongsToMember) {
      res.status(403).json({
        success: false,
        message:
          "You cannot access another member's membership",
      });
      return;
    }

    // Refresh the date-based status, preserving cancellations.
    if (membership.status !== "CANCELLED") {
      const currentStatus = getCurrentStatus(
        membership.startDate,
        membership.expiryDate
      );

      if (membership.status !== currentStatus) {
        membership.status = currentStatus;
        await membership.save();
      }
    }

    res.status(200).json({
      success: true,
      data: membership,
    });
  } catch (error) {
    next(error);
  }
};

// PATCH /api/memberships/:id/cancel
// Admin cancels a membership.
export const cancelMembership = async (
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> => {
  try {
    const { id } = req.params;

    if (!mongoose.isValidObjectId(id)) {
      res.status(400).json({
        success: false,
        message: "Invalid membership ID",
      });
      return;
    }

    const validation = cancelMembershipSchema.safeParse(
      req.body ?? {}
    );

    if (!validation.success) {
      res.status(400).json({
        success: false,
        message: "Validation failed",
        errors: validation.error.flatten(),
      });
      return;
    }

    const membership = await Membership.findById(id);

    if (!membership) {
      res.status(404).json({
        success: false,
        message: "Membership not found",
      });
      return;
    }

    if (membership.status === "CANCELLED") {
      res.status(409).json({
        success: false,
        message: "Membership is already cancelled",
      });
      return;
    }

    const currentStatus = getCurrentStatus(
      membership.startDate,
      membership.expiryDate
    );

    if (currentStatus === "EXPIRED") {
      res.status(409).json({
        success: false,
        message: "Expired memberships cannot be cancelled",
      });
      return;
    }

    membership.status = "CANCELLED";

    await membership.save();

    res.status(200).json({
      success: true,
      message: "Membership cancelled successfully",
      data: {
        membership,
        reason: validation.data.reason ?? null,
      },
    });
  } catch (error) {
    next(error);
  }
};

// POST /api/memberships/:id/renew
// Admin renews a membership by creating a new record.
export const renewMembership = async (
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> => {
  try {
    const { id } = req.params;

    if (!mongoose.isValidObjectId(id)) {
      res.status(400).json({
        success: false,
        message: "Invalid membership ID",
      });
      return;
    }

    const validation = renewMembershipSchema.safeParse(
      req.body ?? {}
    );

    if (!validation.success) {
      res.status(400).json({
        success: false,
        message: "Validation failed",
        errors: validation.error.flatten(),
      });
      return;
    }

    const currentUser = req.user;

    if (!currentUser) {
      res.status(401).json({
        success: false,
        message: "Authentication required",
      });
      return;
    }

    const previousMembership = await Membership.findById(id);

    if (!previousMembership) {
      res.status(404).json({

    }

        message: "A cancelled membership cannot be renewed",

    }

    const planId =
      validation.data.planId ??
      previousMembership.planId.toString();

    const plan = await MembershipPlan.findOne({
      _id: planId,
      isActive: true,
    });

    if (!plan) {
      res.status(404).json({
        success: false,
        message: "Active membership plan not found",
      });
      return;
    }

    const now = new Date();

    // Continue after the previous expiry if it is in the future.
    // Otherwise, start the renewal now.
    const startDate =
      previousMembership.expiryDate > now
        ? new Date(previousMembership.expiryDate)
        : now;

    const expiryDate = new Date(startDate);

    expiryDate.setDate(
      expiryDate.getDate() + plan.durationDays
    );

    // Prevent overlapping active or upcoming memberships.
    const conflictingMembership = await Membership.findOne({
      memberId: previousMembership.memberId,
      status: { $in: ["ACTIVE", "UPCOMING"] },
      startDate: { $lt: expiryDate },
      expiryDate: { $gt: startDate },
    });

    if (conflictingMembership) {
      res.status(409).json({
        success: false,
        message:
          "Another membership already covers the renewal period",
      });
      return;
    }

    const membership = await Membership.create({
      memberId: previousMembership.memberId,
      planId: plan._id,
      startDate,
      expiryDate,
      priceCharged: plan.price,
      status: getCurrentStatus(startDate, expiryDate),
      assignedBy: new mongoose.Types.ObjectId(
        currentUser.userId
      ),
    });

    res.status(201).json({
      success: true,
      message: "Membership renewed successfully",
      data: membership,
    });
  } catch (error) {
    next(error);
  }
};

import { Request, Response, NextFunction } from "express";
import mongoose from "mongoose";

import Payment from "../models/Payment";
import Membership from "../models/Membership";
import { createPaymentSchema } from "../validators/paymentValidator";

// POST /api/payments
// Admin records a payment for an existing membership.
export const createPayment = async (
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> => {
  try {
    const currentUser = req.user;

    if (!currentUser) {
      res.status(401).json({
        success: false,
        message: "Authentication required",
      });
      return;
    }

    if (currentUser.role !== "ADMIN") {
      res.status(403).json({
        success: false,
        message: "Only admins can record payments",
      });
      return;
    }

    const validation = createPaymentSchema.safeParse(req.body);

    if (!validation.success) {
      res.status(400).json({
        success: false,
        message: "Validation failed",
        errors: validation.error.flatten(),
      });
      return;
    }

    const {
      membershipId,
      amount,
      paymentDate,
      paymentMethod,
      reference,
      notes,
    } = validation.data;

    if (!mongoose.isValidObjectId(membershipId)) {
      res.status(400).json({
        success: false,
        message: "Invalid membership ID",
      });
      return;
    }

    if (!Number.isFinite(amount) || amount <= 0) {
      res.status(400).json({
        success: false,
        message: "Payment amount must be greater than zero",
      });
      return;
    }

    const membership = await Membership.findById(membershipId);

    if (!membership) {
      res.status(404).json({
        success: false,
        message: "Membership not found",
      });
      return;
    }

    if (membership.status === "CANCELLED") {
      res.status(409).json({
        success: false,
        message: "Cannot record a payment for a cancelled membership",
      });
      return;
    }

    const actualPaymentDate = paymentDate
      ? new Date(paymentDate)
      : new Date();

    if (Number.isNaN(actualPaymentDate.getTime())) {
      res.status(400).json({
        success: false,
        message: "Invalid payment date",
      });
      return;
    }

    if (actualPaymentDate > new Date()) {
      res.status(400).json({
        success: false,
        message: "Payment date cannot be in the future",
      });
      return;
    }

    // Calculate how much has already been paid for this membership.
    const existingPayments = await Payment.aggregate([
      {
        $match: {
          membershipId: membership._id,
        },
      },
      {
        $group: {
          _id: null,
          totalPaid: { $sum: "$amount" },
        },
      },
    ]);

    const totalPaid = existingPayments[0]?.totalPaid ?? 0;
    const remainingBalance = Math.max(
      0,
      membership.priceCharged - totalPaid
    );

    if (amount > remainingBalance) {
      res.status(409).json({
        success: false,
        message: "Payment exceeds the remaining membership balance",
        data: {
          membershipPrice: membership.priceCharged,
          totalPaid,
          remainingBalance,
        },
      });
      return;
    }

    // Derive the member from the membership, never from client input.
    const payment = await Payment.create({
      memberId: membership.memberId,
      membershipId: membership._id,
      amount,
      paymentDate: actualPaymentDate,
      paymentMethod,
      reference,
      notes,
      recordedBy: new mongoose.Types.ObjectId(currentUser.userId),
    });

    const populatedPayment = await Payment.findById(payment._id)
      .populate("memberId", "name email phone")
      .populate(
        "membershipId",
        "startDate expiryDate priceCharged status"
      )
      .populate("recordedBy", "name email");

    res.status(201).json({
      success: true,
      message: "Payment recorded successfully",
      data: populatedPayment,
    });
  } catch (error) {
    next(error);
  }
};

// PATCH /api/workout-assignments/:id
// Only admins can modify assignment details and status.
export const updateWorkoutAssignment = async (
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> => {
  try {
    const currentUser = req.user;

    if (!currentUser) {
      res.status(401).json({
        success: false,
        message: "Authentication required",
      });
      return;
    }

    if (currentUser.role !== "ADMIN") {
      res.status(403).json({
        success: false,
        message: "Only admins can update workout assignments",
      });
      return;
    }

    const { id } = req.params;

    if (!mongoose.isValidObjectId(id)) {
      res.status(400).json({
        success: false,
        message: "Invalid assignment ID",
      });
      return;
    }

    const updates: unknown = req.body;

    if (
      !updates ||
      typeof updates !== "object" ||
      Array.isArray(updates)
    ) {
      res.status(400).json({
        success: false,
        message: "Request body must be an object",
      });
      return;
    }

    const data = updates as Record<string, unknown>;
    const fields = Object.keys(data);
    const allowedFields = ["status", "endDate", "notes"];

    if (
      fields.length === 0 ||
      fields.some((field) => !allowedFields.includes(field))
    ) {
      res.status(400).json({
        success: false,
        message:
          "Only status, endDate, and notes can be updated",
      });
      return;
    }

    if (
      data.status !== undefined &&
      !["ACTIVE", "COMPLETED", "CANCELLED"].includes(
        String(data.status)
      )
    ) {
      res.status(400).json({
        success: false,
        message: "Invalid assignment status",
      });
      return;
    }

    if (
      data.notes !== undefined &&
      (typeof data.notes !== "string" || data.notes.length > 1000)
    ) {
      res.status(400).json({
        success: false,
        message: "Notes must be text of at most 1000 characters",
      });
      return;
    }

    let parsedEndDate: Date | null | undefined;

    if (data.endDate === null) {
      parsedEndDate = null;
    } else if (data.endDate !== undefined) {
      if (typeof data.endDate !== "string") {
        res.status(400).json({
          success: false,
          message: "End date must be a valid date or null",
        });
        return;
      }

      parsedEndDate = new Date(data.endDate);

      if (Number.isNaN(parsedEndDate.getTime())) {
        res.status(400).json({
          success: false,
          message: "Invalid end date",
        });
        return;
      }
    }

    const assignment = await WorkoutAssignment.findById(id);

    if (!assignment) {
      res.status(404).json({
        success: false,
        message: "Assignment not found",
      });
      return;
    }

    const effectiveEndDate =
      parsedEndDate === undefined
        ? assignment.endDate
        : parsedEndDate;

    if (
      effectiveEndDate &&
      effectiveEndDate < assignment.startDate
    ) {
      res.status(400).json({
        success: false,
        message: "End date must be on or after the start date",
      });
      return;
    }

    if (data.status !== undefined) {
      assignment.status = data.status as typeof assignment.status;
    }

    if (parsedEndDate !== undefined) {
      assignment.endDate = parsedEndDate as typeof assignment.endDate;
    }

    if (data.notes !== undefined) {
      assignment.notes = (data.notes as string).trim();
    }

    await assignment.save();

    await assignment.populate([
      { path: "memberId", select: "name email" },
      { path: "workoutPlanId" },
      { path: "assignedBy", select: "name email" },

function uploadToCloudinary(
  fileBuffer: Buffer
): Promise<{ secure_url: string; public_id: string }> {
  return new Promise((resolve, reject) => {
    const stream = cloudinary.uploader.upload_stream(
      {
        folder: "fittrack/workout-images",
        resource_type: "image",
      },
      (error, result) => {
        if (error || !result) {
          reject(error ?? new Error("Cloudinary upload failed"));
          return;
        }

        resolve({
          secure_url: result.secure_url,
          public_id: result.public_id,
        });
      }
    );

    stream.end(fileBuffer);
  });
}

export async function uploadWorkoutPlanImage(
  req: Request,
  res: Response
): Promise<void> {
  let newPublicId: string | undefined;
  let databaseSaveSucceeded = false;

  try {
    const { id } = req.params;

    // Validate the workout plan ID.
    if (!mongoose.isValidObjectId(id)) {
      res.status(400).json({
        success: false,
        message: "Invalid workout plan ID.",
      });
      return;
    }

    // Ensure an image was uploaded.
    if (!req.file) {
      res.status(400).json({
        success: false,
        message: "Please select a workout image.",
      });
      return;
    }

    // Find the workout plan.
    const workoutPlan = await WorkoutPlan.findById(id);

    if (!workoutPlan) {
      res.status(404).json({
        success: false,
        message: "Workout plan not found.",
      });
      return;
    }

    // Upload the new image to Cloudinary.
    const uploadedImage = await uploadToCloudinary(req.file.buffer);
    newPublicId = uploadedImage.public_id;

    // Keep the old image details until the database save succeeds.
    const oldPublicId = workoutPlan.imagePublicId;

    workoutPlan.imageUrl = uploadedImage.secure_url;
    workoutPlan.imagePublicId = uploadedImage.public_id;

    await workoutPlan.save();
    databaseSaveSucceeded = true;

    // Delete the previous image only after the new image is saved.
    if (oldPublicId && oldPublicId !== newPublicId) {
      try {
        await cloudinary.uploader.destroy(oldPublicId);
      } catch (deleteError) {
        console.error(
          "Could not delete previous workout image:",
          deleteError
        );
      }
    }

    res.status(200).json({
      success: true,
      message: "Workout image uploaded successfully.",
      data: {
        workoutPlanId: workoutPlan._id,
        imageUrl: workoutPlan.imageUrl,
      },
    });
  } catch (error) {
    console.error("Workout image upload error:", error);

    // Clean up the new image only if the database save did not succeed.
    // If the save outcome is uncertain due to a network error, avoid
    // deleting an image that the database might already reference.
    if (newPublicId && !databaseSaveSucceeded) {
      try {
        await cloudinary.uploader.destroy(newPublicId);
      } catch (cleanupError) {
        console.error(
          "Could not clean up uploaded workout image:",
          cleanupError
        );
      }
    }

    res.status(500).json({
      success: false,
      message: "Unable to upload workout image.",
    });
  }
}

import type { Request, Response, NextFunction } from "express";

import { verifyToken } from "../utils/jwt";
import User from "../models/User";

export const authenticate = async (
  req: Request,
  res: Response,
  next: NextFunction
): Promise<void> => {
  const token = req.cookies?.fittrack_token;

  // 1. Require the authentication cookie.
  if (!token || typeof token !== "string") {
    res.status(401).json({
      success: false,
      message: "Authentication required.",
    });
    return;
  }

  // 2. Verify the JWT.
  let payload: ReturnType<typeof verifyToken>;

  try {
    payload = verifyToken(token);
  } catch {
    res.status(401).json({
      success: false,
      message: "Invalid or expired authentication token.",
    });
    return;
  }

  // 3. Confirm the user still exists and is active.
  try {
    const user = await User.findById(payload.userId)
      .select("_id role isActive");

    if (!user || !user.isActive) {
      res.status(401).json({
        success: false,
        message: "Account not found or inactive.",
      });
      return;
    }

    // 4. Attach trusted user details to the request.
    req.user = {
      userId: user._id.toString(),
      role: user.role,
    };

    next();
  } catch (error) {
    next(error);
  }
};

import type { Request, Response, NextFunction } from "express";

export const errorMiddleware = (
  error: unknown,
  _req: Request,
  res: Response,
  _next: NextFunction
): void => {
  // Do not attempt to send a second response.
  if (res.headersSent) {
    return;
  }

  // Handle MongoDB duplicate-key errors.
  if (
    typeof error === "object" &&
    error !== null &&
    "code" in error &&
    error.code === 11000
  ) {
    res.status(409).json({
      success: false,
      message: "A record with this value already exists.",
    });
    return;
  }

  // Log the full error on the server, not in the client response.
  console.error("Unhandled application error:", error);

  res.status(500).json({
    success: false,
    message: "Internal server error.",
  });
};

  "image/jpeg",
  "image/png",
  "image/webp",
];

const storage = multer.memoryStorage();

const upload = multer({
  storage,
  limits: {
    fileSize: 5 * 1024 * 1024, // Maximum 5 MB
    files: 1, // One image per request
  },
  fileFilter: (
    _req: Request,
    file: Express.Multer.File,
    callback: multer.FileFilterCallback
  ) => {
    if (!allowedMimeTypes.includes(file.mimetype)) {
      callback(
        new Error("Only JPEG, PNG, and WebP images are allowed.")
      );
      return;
    }

    callback(null, true);
  },
});

// Handle errors produced by the upload middleware.
// Mount this after the routes that use the upload middleware.
export function handleUploadError(
  error: unknown,
  _req: Request,
  res: Response,
  next: NextFunction
): void {
  if (error instanceof multer.MulterError) {
    if (error.code === "LIMIT_FILE_SIZE") {
      res.status(400).json({
        success: false,
        message: "Image size must not exceed 5 MB.",
      });
      return;
    }

    if (error.code === "LIMIT_FILE_COUNT") {
      res.status(400).json({
        success: false,
        message: "Only one image can be uploaded per request.",
      });
      return;
    }

    res.status(400).json({
      success: false,
      message: "Image upload failed. Please check the uploaded file.",
    });
    return;
  }

  if (
    error instanceof Error &&
    error.message ===
      "Only JPEG, PNG, and WebP images are allowed."
  ) {
    res.status(400).json({
      success: false,
      message: error.message,
    });
    return;

}

function validateBody(schema: { safeParse: (data: unknown) => any }) {
  return (req: Request, res: Response, next: NextFunction): void => {
    const result = schema.safeParse(req.body);

    if (!result.success) {
      res.status(400).json({
        success: false,
        message: "Validation failed.",
        errors: result.error.issues,
      });
      return;
    }

    req.body = result.data;
    next();
  };
}

import jwt, { type SignOptions } from "jsonwebtoken";
import mongoose from "mongoose";
import type { UserRole } from "../models/User";

const getJwtSecret = (): string => {
  const secret = process.env.JWT_SECRET;

  if (!secret || secret.length < 32) {
    throw new Error(
      "JWT_SECRET must be set to a secret of at least 32 characters."
    );
  }

  return secret;
};

export interface TokenPayload {
  userId: string;
  role: UserRole;
}

const isUserRole = (role: unknown): role is UserRole =>
  role === "ADMIN" || role === "MEMBER";

export const createToken = (payload: TokenPayload): string => {
  const options: SignOptions = {
    expiresIn: "1d",
  };

  return jwt.sign(payload, getJwtSecret(), options);
};

export const verifyToken = (token: string): TokenPayload => {
  const decoded = jwt.verify(token, getJwtSecret());

  if (
    typeof decoded === "string" ||
    !mongoose.isValidObjectId(decoded.userId) ||
    !isUserRole(decoded.role)
  ) {
    throw new Error("Invalid token payload.");
  }

  return {
    userId: decoded.userId,
    role: decoded.role,
  };
};
